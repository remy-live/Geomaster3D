class Cylinder extends Solid { constructor(c, rx, ry, h, o, i) { super(); this.c = { x: c.x, y: c.y }; this.rx = rx; this.ry = ry; this.h = { x: h.x, y: h.y }; this.orient = o; this.inverted = i; } invert() { this.inverted = !this.inverted; } renderHandles(g) { g.appendChild(Utils.createEl("circle", { cx: this.c.x, cy: this.c.y, r: 5, class: "handle", "data-idx": 0, "data-sid": this.id })); g.appendChild(Utils.createEl("circle", { cx: this.c.x + this.rx, cy: this.c.y, r: 5, class: "handle", "data-idx": 1, "data-sid": this.id })); g.appendChild(Utils.createEl("circle", { cx: this.c.x, cy: this.c.y + this.ry, r: 5, class: "handle", "data-idx": 2, "data-sid": this.id })); g.appendChild(Utils.createEl("circle", { cx: this.c.x + this.h.x, cy: this.c.y + this.h.y, r: 5, class: "handle", "data-idx": 3, "data-sid": this.id })); } moveHandle(i, p) { i = parseInt(i); if (i === 0) this.c = p; else if (i === 1) this.rx = Utils.dist(this.c, p); else if (i === 2) this.ry = Utils.dist(this.c, p); else if (i === 3) { this.h = { x: p.x - this.c.x, y: p.y - this.c.y }; } } render(group, l, idx) { const g = Utils.createEl("g", { "data-id": this.id, class: "solid-group" }); g.style.setProperty('--stroke-color', this.color); if (this.selected) g.classList.add("selected"); const p1 = { x: this.c.x, y: this.c.y }; const p2 = { x: this.c.x + this.h.x, y: this.c.y + this.h.y }; const rx = this.rx, ry = this.ry; // 1. Calcul des points de tangence (bords du cylindre) const { t1, t2 } = Utils.getEllipseTangents(rx, ry, this.h.x, this.h.y); const x1 = rx * Math.cos(t1), y1 = ry * Math.sin(t1); const x2 = rx * Math.cos(t2), y2 = ry * Math.sin(t2); // 2. Zone cliquable (Hitbox) g.appendChild(Utils.createEl("ellipse", { cx: p1.x, cy: p1.y, rx, ry, class: "face face-base" })); const polyPts = `${p1.x + x1},${p1.y + y1} ${p1.x + x2},${p1.y + y2} ${p2.x + x2},${p2.y + y2} ${p2.x + x1},${p2.y + y1}`; const hit = Utils.createEl("polygon", { points: polyPts, class: "face" }); g.appendChild(hit); // 3. Dessin des bords (Arêtes latérales) g.appendChild(Utils.createEl("line", { x1: p1.x + x1, y1: p1.y + y1, x2: p2.x + x1, y2: p2.y + y1, class: "edge-solid" })); g.appendChild(Utils.createEl("line", { x1: p1.x + x2, y1: p1.y + y2, x2: p2.x + x2, y2: p2.y + y2, class: "edge-solid" })); // 4. Définition des bases (P1 = Start, P2 = End) // Si inversé, P1 est plein, P2 est coupé. Sinon l'inverse. const solidBaseCenter = this.inverted ? p1 : p2; const splitBaseCenter = this.inverted ? p2 : p1; // 5. Dessin de la base PLEINE (toujours un cercle complet visible) g.appendChild(Utils.createEl("ellipse", { cx: solidBaseCenter.x, cy: solidBaseCenter.y, rx, ry, class: "edge-solid" })); // 6. Dessin de la base COUPÉE (Demi-pointillé / Demi-plein) // On calcule le point médian d'un des arcs pour savoir s'il pointe vers l'autre base const midAngle = (t1 + t2) / 2; // Vecteur du centre vers ce point médian const vMidX = Math.cos(midAngle); const vMidY = Math.sin(midAngle); // Direction de la hauteur (vers l'autre base ou s'en éloignant ?) // Si on est sur P1, la hauteur va vers P2. Si on est sur P2, la hauteur "interne" va vers P1 (donc -h). const hDirX = this.inverted ? -this.h.x : this.h.x; const hDirY = this.inverted ? -this.h.y : this.h.y; // Produit scalaire : Si > 0, l'arc pointe dans la même direction que le corps du cylindre => C'est l'intérieur (Pointillé) // Note: ellipse scaling (rx, ry) affecte le vecteur normal visuel, mais le signe du produit scalaire reste bon pour la direction générale. const dotProd = (vMidX * rx * hDirX) + (vMidY * ry * hDirY); // Points de départ et fin des arcs sur la base coupée const ptT1 = { x: splitBaseCenter.x + x1, y: splitBaseCenter.y + y1 }; const ptT2 = { x: splitBaseCenter.x + x2, y: splitBaseCenter.y + y2 }; // Construction des chemins SVG // Arc A : de T1 à T2 // Arc B : de T2 à T1 const pathA = `M ${ptT1.x} ${ptT1.y} A ${rx} ${ry} 0 0 1 ${ptT2.x} ${ptT2.y}`; const pathB = `M ${ptT2.x} ${ptT2.y} A ${rx} ${ry} 0 0 1 ${ptT1.x} ${ptT1.y}`; // Si le produit scalaire est positif, l'Arc A ("milieu" entre t1 et t2) est à l'intérieur if (dotProd > 0) { g.appendChild(Utils.createEl("path", { d: pathA, class: "edge-dashed", fill: "none" })); // Intérieur g.appendChild(Utils.createEl("path", { d: pathB, class: "edge-solid", fill: "none" })); // Extérieur } else { g.appendChild(Utils.createEl("path", { d: pathB, class: "edge-dashed", fill: "none" })); // Intérieur g.appendChild(Utils.createEl("path", { d: pathA, class: "edge-solid", fill: "none" })); // Extérieur } group.appendChild(g); return idx; } getData() { return { type: 'cylinder', c: this.c, rx: this.rx, ry: this.ry, h: this.h, orient: this.orient, inv: this.inverted, color: this.color }; } }
